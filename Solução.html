'); return escaped.replace(/\[\[(.*?)\]\]/g, function(_, name) { const safeName = String(name || '').trim(); if (!/^[\w\-\.\s\/]+$/.test(safeName)) return escapeHtml('[[' + safeName + ']]'); const url = docsBase + encodeURIComponent(safeName).replace(/%2F/g, '/'); const label = escapeHtml(safeName); return `${label}`; }); } function buildObsElement(obsText, docsBase) { const container = document.createElement('div'); container.className = 'obs'; if (!obsText) return container; docsBase = (docsBase || '').replace(/\/+$/, '') + '/'; obsText = String(obsText).replace(/\[\[br\]\]/gi, '\n'); const parts = obsText.split(/\[\[(.*?)\]\]/g); parts.forEach((part, i) => { if (i % 2 === 1) { const safeName = part.trim(); if (/^[\w\-\.\s\/]+$/.test(safeName)) { const a = document.createElement('a'); a.href = docsBase + encodeURIComponent(safeName).replace(/%2F/g, '/'); a.target = '_blank'; a.rel = 'noopener noreferrer'; a.className = 'doc-link'; a.textContent = safeName; container.appendChild(a); } else { container.appendChild(document.createTextNode('[[' + safeName + ']]')); } } else { const lines = part.split('\n'); for (let li = 0; li < lines.length; li++) { if (li) container.appendChild(document.createElement('br')); container.appendChild(document.createTextNode(lines[li])); } } }); return container; }
